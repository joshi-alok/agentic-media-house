**Create `tsconfig.base.json`:**
```json
{
  "compilerOptions": {
    "target": "ES2023",
    "lib": ["ES2023"],
    "module": "ESNext",
    "moduleResolution": "bundler",
    "types": ["node"],
    "strict": true,
    "noEmit": true,
    "skipLibCheck": true,
    "isolatedModules": true,
    "verbatimModuleSyntax": true
  }
}
```

Why these options (TypeScript 7's `tsc` is the native compiler and is stricter about configuration than older versions):
- `moduleResolution: "bundler"` lets source files import each other without file extensions, which is how Vitest, tsx and Astro all resolve them. The older `node` setting no longer exists in TypeScript 7.
- `types: ["node"]` is required. TypeScript 7 no longer loads every installed `@types/*` package on its own, so without it `import { readFileSync } from 'node:fs'` fails to typecheck.
- `verbatimModuleSyntax` makes a type-only import without the `type` keyword an error, which keeps imports honest under `isolatedModules`.
- `noEmit` because nothing is compiled: `tsc` only checks.

The schemas describe shape and format: ids, hashes, dates, URLs, enums, lengths. Whether the content is acceptable (a fact sentence cites a claim, the cited claim exists, a category is in the vocabulary, the title is not empty) is the gate's job in a later milestone, so that the gate can report it by code and send the failure into a revision round instead of crashing on a schema error. Where a rule is deliberately left to the gate, the schema file says so and a test locks that in.

- [ ] **Step 1: Write the test support file**

These helpers keep the schema tests short and make each failure name the exact field that was rejected.

**Create `gate/test/schema-helpers.ts`:**
```ts
import { expect } from 'vitest';
import type { z } from 'zod';

/** One line per validation issue, as `dotted.path: message`. The root path is `(root)`. */
export function issueLines(schema: z.ZodType, value: unknown): string[] {
  const result = schema.safeParse(value);
  if (result.success) return [];
  return result.error.issues.map((issue) => `${issue.path.join('.') || '(root)'}: ${issue.message}`);
}

/** The dotted paths of every failing field. An empty list means the value is valid. */
export function failedPaths(schema: z.ZodType, value: unknown): string[] {
  return issueLines(schema, value).map((line) => line.slice(0, line.indexOf(': ')));
}

/** Assert that `value` parses, printing the validation issues if it does not. */
export function expectValid(schema: z.ZodType, value: unknown): void {
  const result = schema.safeParse(value);
  expect(result.success ? [] : result.error.issues).toEqual([]);
}

/**
 * Return a deep copy of `base` with the value at the dotted `path` replaced
 * (array indexes are path segments: `sections.0.heading`). Passing `undefined`
 * makes a required field look missing. The result is `unknown` on purpose: tests
 * use this to build values that are NOT valid, which the types would forbid.
 */
export function patch(base: unknown, path: string, value: unknown): unknown {
  const copy = structuredClone(base) as Record<string, unknown>;
  const keys = path.split('.');
  const last = keys[keys.length - 1] as string;
  let cursor = copy;
  for (const key of keys.slice(0, -1)) {
    cursor = cursor[key] as Record<string, unknown>;
  }
  cursor[last] = value;
  return copy;
}
```

- [ ] **Step 7: Write the schemas**

Zod 4 notes: `z.iso.date()` and `z.iso.datetime()` reject impossible dates such as 2026-02-30; `z.url({ protocol })` restricts the scheme; `z.strictObject` rejects unknown keys; `z.record(z.enum(CHECK_IDS), value)` requires every key of the enum to be present. A refinement attached to an object (`.refine(...)`) still nests inside a larger `z.object`. Each schema and its inferred type share one name (`export const Claim` and `export type Claim`), so `import { Claim } from '@robywire/gate'` gives both.

**Create `gate/src/schemas.ts`:**
```ts
import { z } from 'zod';
import {
  BASES,
  CATEGORY_VOCAB,
  CHECK_IDS,
  CHECK_RESULTS,
  CLAIM_TYPES,
  CORRECTION_KINDS,
  LEDGER_SECTIONS,
  ROUTES,
  RUN_OUTCOMES,
  SENTENCE_KINDS,
  SOURCE_STATUSES,
  STATUSES,
  VALUE_KINDS,
} from './constants';

// Schemas describe SHAPE and FORMAT: ids, hashes, dates, URLs, enums, lengths.
// Whether the CONTENT is acceptable (a fact sentence cites a claim, the claim
// exists, the category is in the vocabulary, the title is not empty) is the
// gate's job. That way the gate reports it by code and the failure goes into
// a revision round, instead of dying as an unreadable schema error.

const text = z.string().min(1);
const httpUrl = z.url({ protocol: /^https?$/ });
const isoDate = z.iso.date();
const isoDateTime = z.iso.datetime({ offset: true });

/** "c-12". Claim ids are assigned by the extractor and cited by sentences. */
export const ClaimId = z.string().regex(/^c-\d+$/, 'claim id must look like c-12');

// --- Claims -----------------------------------------------------------------

export const ClaimValue = z.object({
  kind: z.enum(VALUE_KINDS),
  raw: text,
  normalised: text,
  unit: text.optional(),
});
export type ClaimValue = z.infer<typeof ClaimValue>;

export const ClaimSource = z
  .object({
    url: httpUrl,
    tier: z.union([z.literal(1), z.literal(2)]),
    snapshotHash: z.string().regex(/^[0-9a-f]{64}$/, 'snapshotHash must be 64 lowercase hex characters'),
    retrievedAt: isoDateTime,
    outlet: text.optional(),
  })
  .refine((source) => source.tier !== 2 || source.outlet !== undefined, {
    message: 'outlet is required when tier is 2',
    path: ['outlet'],
  });
export type ClaimSource = z.infer<typeof ClaimSource>;

export const ClaimSpan = z
  .object({
    start: z.number().int().nonnegative(),
    end: z.number().int().nonnegative(),
    quote: z.string().min(1).max(300),
  })
  .refine((span) => span.end > span.start, {
    message: 'span.end must be greater than span.start',
    path: ['end'],
  });
export type ClaimSpan = z.infer<typeof ClaimSpan>;

export const Claim = z.object({
  id: ClaimId,
  text,
  label: text,
  type: z.enum(CLAIM_TYPES),
  source: ClaimSource,
  span: ClaimSpan,
  basis: z.enum(BASES),
  status: z.enum(STATUSES),
  values: z.array(ClaimValue),
});
export type Claim = z.infer<typeof Claim>;

// --- Sentences and drafts ---------------------------------------------------

// `claimIds` on a fact sentence may be empty here on purpose: G3 reports it.
export const FactSentence = z.object({
  kind: z.literal('fact'),
  text: z.string(),
  claimIds: z.array(z.string()),
  attributed: z.boolean().optional(),
});
export type FactSentence = z.infer<typeof FactSentence>;

export const AnalysisSentence = z.object({
  kind: z.literal('analysis'),
  text: z.string(),
  claimIds: z.array(z.string()),
});
export type AnalysisSentence = z.infer<typeof AnalysisSentence>;

// A limitation describes exactly one claim's metadata, through one template.
export const LimitationSentence = z.object({
  kind: z.literal('limitation'),
  text: z.string(),
  claimIds: z.tuple([z.string()]),
  template: z.enum(['basis', 'status']),
});
export type LimitationSentence = z.infer<typeof LimitationSentence>;

export const Sentence = z.discriminatedUnion('kind', [FactSentence, AnalysisSentence, LimitationSentence]);
export type Sentence = z.infer<typeof Sentence>;

export const Draft = z.object({
  slug: z.string().regex(/^[a-z0-9]+(?:-[a-z0-9]+)*$/, 'slug must be lowercase kebab-case'),
  publishedAt: isoDate,
  // A list of strings only. G1 checks the slugs against CATEGORY_VOCAB.
  categories: z.array(z.string()),
  author: z.object({ name: z.string() }),
  title: Sentence,
  description: Sentence,
  opening: z.array(Sentence),
  takeaways: z.array(Sentence),
  sections: z.array(
    z.object({
      heading: z.string(),
      paragraphs: z.array(z.array(Sentence)),
    }),
  ),
  limitations: z.array(LimitationSentence),
});
export type Draft = z.infer<typeof Draft>;

// --- Ledger -----------------------------------------------------------------

export const LedgerSentence = z.object({
  index: z.number().int().nonnegative(),
  section: z.enum(LEDGER_SECTIONS),
  heading: z.string().optional(),
  kind: z.enum(SENTENCE_KINDS),
  text: z.string(),
  claimIds: z.array(z.string()),
});
export type LedgerSentence = z.infer<typeof LedgerSentence>;

export const LedgerEntry = z.object({
  schemaVersion: z.literal(1),
  slug: z.string().min(1),
  runId: text,
  publishedAt: isoDate,
  title: z.string().min(1),
  categories: z.array(z.string()),
  claims: z.array(Claim),
  // Claim ids in the order the body first cites them.
  order: z.array(ClaimId),
  sentences: z.array(LedgerSentence),
  gate: z.object({
    route: z.enum(ROUTES),
```

```ts
function swap(text: string, from: string, to: string): string {
  expect(text).toContain(from);

/** The dotted path of every validation issue `parseSources` reports for `yamlText`. */
function failurePaths(yamlText: string): string[] {
  try {
    parseSources(yamlText);
  } catch (error) {
    if (error instanceof ZodError) {
      return error.issues.map((issue) => issue.path.join('.') || '(root)');
    }
    throw error;
  }
```

**Create `gate/src/sources.ts`:**
```ts
import { readFileSync } from 'node:fs';
import { parse } from 'yaml';
import { SourcesRegistry } from './schemas';

/**
 * Parse the text of sources.yml and validate it against `SourcesRegistry`.
 * Throws a YAMLParseError for text that is not valid YAML (including duplicate
 * keys) and a ZodError for YAML that is not a valid registry.
 */
export function parseSources(yamlText: string): SourcesRegistry {
  return SourcesRegistry.parse(parse(yamlText));
}

/** Read the file at `path` and validate it with `parseSources`. */
export function loadSources(path: string): SourcesRegistry {
  return parseSources(readFileSync(path, 'utf8'));
}
```
